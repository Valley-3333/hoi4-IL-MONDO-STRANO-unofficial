drunks = {
    possible = {
        # classic triggers used in all vanilla achievements
        difficulty > 1
        #has_start_date < 1936.01.02
        game_rules_allow_achievements = yes
		#achievment specific
		OR = {
		    original_tag = FIN
            original_tag = EST
		}
    }
    happened = {
        custom_trigger_tooltip = {
            tooltip = EST_FIN_or_UGRA_trigger_tt
            OR = {
                has_cosmetic_tag = EST_FIN
                has_cosmetic_tag = EST_UGRA
            }
        }
        has_offensive_war_with = SOV
        SOV = {
            controls_province = 3152
            controls_province = 11105
        }
    }
}

tsuyosa_shoumei = {
    possible = {
        difficulty > 1
        game_rules_allow_achievements = yes
        original_tag = ALB
    }
    happened = {
        ALB = {
            is_exists = yes
            is_in_faction = no
            is_puppet = no
            NOT = {
                any_owned_state = {
                    any_province_building_level = {
                        building = bunker
                        level < 10
                    }
                }
            }
            date > 1945.01.01

        }
    }
}
