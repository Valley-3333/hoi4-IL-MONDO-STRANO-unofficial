chile_chirijiri = {
	possible = {
		difficulty > 1
		game_rules_allow_achievements = yes
	}
	happened = {
		
		# 1ヶ国目
		any_country = {
			any_owned_state = { is_core_of = CHL }
			
			# 2ヶ国目
			any_country = {
				NOT = { tag = PREV } # 1ヶ国目とは違う国
				any_owned_state = { is_core_of = CHL }
				
				# 3ヶ国目
				any_country = {
					NOT = { tag = PREV }      # 2ヶ国目とは違う国
					NOT = { tag = PREV.PREV } # 1ヶ国目とも違う国
					any_owned_state = { is_core_of = CHL }

					any_country = {
						NOT = { tag = PREV }      # 3ヶ国目とは違う国
						NOT = { tag = PREV.PREV } # 2ヶ国目とも違う国
						NOT = { tag = PREV.PREV.PREV } # 1ヶ国目とも違う国
						any_owned_state = { is_core_of = CHL }
					}
				}
			}
		}
	}
}
