scripted_gui = {

	mnds_propaganda_window = {
		window_name = "mnds_propaganda_window"
		context_type = player_context

		visible = {
			has_country_flag = Propaganda_Visible
		}

		triggers = {
			gakubuchi_for_init_visible = {
				has_country_flag = mnds_init
			}
			gakubuchi_for_pic_1_visible = {
				has_country_flag = mnds_pic_1
			}
			gakubuchi_for_sinesine_visible = {
				has_country_flag = mnds_sinesine_call
			}

			mnds_init_visible = {
				has_country_flag = mnds_init
			}
			link_button_1_visible = {
				has_country_flag = mnds_init
			}
			link_button_2_visible = {
				has_country_flag = mnds_pic_1
			}
			mnds_sinesine_call_visible = {
				has_country_flag = mnds_sinesine_call
			}
			mnds_sinesine_call_link_visible = {
				has_country_flag = mnds_sinesine_call
			}
		}
		############
		###BUTTON###
		############

		effects = {
			close_button_click = {
				clr_country_flag = Propaganda_Visible

				if = {
					limit = {
						has_country_flag = mnds_init
					}
					clr_country_flag = mnds_init
				}
				if = {
					limit = {
						has_country_flag = mnds_pic_1
					}
					clr_country_flag = mnds_pic_1
				}
				if = {
					limit = {
						has_country_flag = mnds_sinesine_call
					}
					clr_country_flag = mnds_sinesine_call

				}

		}
